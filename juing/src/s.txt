/* ── Art Deco fan ornament ──────────────────────────────── */
function Fan({ width = 96, opacity = 1, light = false }) {
  const color = light ? 'rgba(255,255,255,0.6)' : 'rgba(0,0,0,0.4)';
  const h = width / 2;
  const cx = width / 2;
  const cy = h;
  const R  = h - 1;
  return (
    <svg viewBox={`0 0 ${width} ${h}`} width={width} height={h} fill="none" aria-hidden style={{ opacity }}>
      {Array.from({ length: 9 }, (_, i) => {
        const a = (Math.PI * i) / 8;
        return (
          <line key={i} x1={cx} y1={cy}
            x2={cx + R * Math.cos(Math.PI - a)} y2={cy - R * Math.sin(a)}
            stroke={color} strokeWidth="0.6" />
        );
      })}
      {[0.88, 0.63, 0.37].map((f, i) => (
        <path key={i}
          d={`M${cx - R * f} ${cy} A${R * f} ${R * f} 0 0 1 ${cx + R * f} ${cy}`}
          stroke={color} strokeWidth="0.6" fill="none" />
      ))}
      <line x1={0} y1={cy} x2={width} y2={cy} stroke={color} strokeWidth="0.6" opacity="0.35" />
    </svg>
  );
}

/* ── Corner bracket ─────────────────────────────────────── */
function Corner({ pos, light = false }) {
  const t  = pos.includes('top')   ? 'top-6'    : 'bottom-6';
  const l  = pos.includes('left')  ? 'left-6'   : 'right-6';
  const bT = pos.includes('top')   ? 'border-t' : 'border-b';
  const bL = pos.includes('left')  ? 'border-l' : 'border-r';
  const cl = light ? 'border-white/30' : 'border-black/20';
  return <div className={`absolute ${t} ${l} w-12 h-12 ${bT} ${bL} ${cl}`} />;
}
